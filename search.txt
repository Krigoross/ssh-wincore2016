index=_internal source=*splunkd.log* earliest=-24h latest=now
("Connected to idx" OR "Cooked connection"
 OR "Connection to host" OR "Connect to")
| eval resultat=case(
    match(_raw,"(?i)\bConnected to idx"),"reussite",
    match(_raw,"(?i)\bCooked connection.*timed out"),"echec",
    match(_raw,"(?i)\bConnection to host.*failed"),"echec",
    match(_raw,"(?i)\bConnect to .*failed"),"echec"
  )
| where isnotnull(resultat)
| rex field=_raw "(?i)\b(?:idx|ip|host_dest)=(?<destination>[^,\s;]+)"
| eval destination=coalesce(destination,"Non extraite : voir exemple")
| stats count(eval(resultat="reussite")) AS reussites
        count(eval(resultat="echec")) AS echecs
        latest(_raw) AS exemple
  BY host destination
| sort 0 - echecs
