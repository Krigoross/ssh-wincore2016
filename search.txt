| tstats summariesonly=false count
    min(_time) as firstTime
    max(_time) as lastTime
  from datamodel=Endpoint.Processes
  where earliest=-15m latest=now
    (
      Processes.parent_process_name="php*"
      OR Processes.parent_process_name="apache2"
      OR Processes.parent_process_name="httpd"
    )
    (
      Processes.process_name IN (
        "sh","bash","dash","ash","zsh","ksh",
        "curl","wget","nc","ncat","netcat","socat",
        "id","whoami","uname"
      )
      OR Processes.process_name="python*"
      OR Processes.process_name="perl*"
    )
  by Processes.dest Processes.user
     Processes.parent_process_name
     Processes.process_name Processes.process
| rename Processes.* AS *
| convert ctime(firstTime) ctime(lastTime)
| sort - lastTime
