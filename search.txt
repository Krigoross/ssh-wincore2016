index=_internal source=*splunkd.log* earliest=-24h latest=now
("Connected to idx" OR "Cooked connection"
 OR "Connection to host" OR "Connect to")
| eval resultat=case(
    match(_raw,"(?i)\bConnected to idx"),"reussite",
    match(_raw,"(?i)\bCooked connection.*timed out"),"echec",
    match(_raw,"(?i)\bConnection to host.*failed"),"echec",
    match(_raw,"(?i)\bConnect to .*failed"),"echec"
  )
| where isnotnull(resultat)
| stats count(eval(resultat="reussite")) AS nb_reussites
        count(eval(resultat="echec")) AS nb_echecs
        max(eval(if(resultat="reussite",_time,null()))) AS derniere_reussite
        max(eval(if(resultat="echec",_time,null()))) AS dernier_echec
  BY host
| eval groupe=case(
    nb_reussites>0 AND nb_echecs>0,"Réussites et échecs",
    nb_echecs>0,"Échecs uniquement",
    true(),"Réussites uniquement"
  )
| convert ctime(derniere_reussite) ctime(dernier_echec)
| table groupe host nb_reussites nb_echecs derniere_reussite dernier_echec
| sort 0 groupe - nb_echecs
