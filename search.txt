index=_internal source=*metrics.log* group=queue
host="splunk-idx-01" name="parsingqueue"
earliest=-24h latest=now
| eval remplissage_pct=case(
    max_size_kb>0,100*current_size_kb/max_size_kb,
    max_size>0,100*current_size/max_size
  )
| timechart span=5m
    avg(remplissage_pct) AS remplissage_moyen_pct
    max(remplissage_pct) AS remplissage_max_pct
    count AS nb_mesures
    count(eval(blocked="true")) AS mesures_bloquees
| eval pct_mesures_bloquees=round(100*mesures_bloquees/nb_mesures,1)
