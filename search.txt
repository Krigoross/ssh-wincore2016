index=_internal source=*metrics.log* group=queue earliest=-24h latest=now
| eval remplissage_pct=case(
    max_size_kb>0,100*current_size_kb/max_size_kb,
    max_size>0,100*current_size/max_size
  )
| stats max(remplissage_pct) AS remplissage_max_pct
        count(eval(blocked="true")) AS mesures_bloquees
  BY host name
| eval remplissage_max_pct=round(remplissage_max_pct,1)
| sort 0 - mesures_bloquees - remplissage_max_pct
